param(
  [switch]$QuotaVerified,
  [string]$QuotaEvidence
)
$ErrorActionPreference = 'Stop'
if (!$QuotaVerified.IsPresent -or [string]::IsNullOrWhiteSpace($QuotaEvidence) -or $args.Count -ne 0) {
  throw 'Fresh exact qwen3.7-plus free-quota row and stop-when-exhausted setting must be verified and described before this one-shot diagnostic.'
}
$taskFolder = 'C:/Users/LOCAL_USER/AppData/Local/Iris/bailian-eval-20260918'
$taskRoot = 'D:/work/AGE-org/.worktrees/iris-daily-pilot-1eb86'
foreach ($taskMarker in @('identity-target-diagnostic-20260925.json',
  'identity-target-diagnostic-20260925.started.json',
  'identity-target-diagnostic-20260925.halted.json')) {
  if (Test-Path -LiteralPath (Join-Path $taskFolder $taskMarker)) {
    throw 'Existing identity-target diagnostic artifact; no overwrite or repeat.'
  }
}
$taskPrevious = Join-Path $taskFolder 'literal-receipt-diagnostic-20260925.json'
if (!(Test-Path -LiteralPath $taskPrevious) -or
  (Get-FileHash -Algorithm SHA256 -LiteralPath $taskPrevious).Hash.ToLowerInvariant() -ne
    '5e40ecbe2f7618655b5a9372eb4928fbfb02f6d43353c4808d81d7c091ccb830') {
  throw 'Retained failed literal-receipt diagnostic is unavailable or changed.'
}
$taskSources = @(
  'apps/core/src/config/env.ts',
  'apps/core/src/model/openai-compatible-chat-completions-client.ts',
  'apps/core/src/proactive-discussion/model.ts',
  'apps/core/src/proactive-discussion/review-receipts.ts',
  'scripts/pilot/proactive-discussion-eval.ts'
)
$taskDirty = & git -C $taskRoot status --porcelain -- $taskSources
if ($LASTEXITCODE -ne 0 -or $taskDirty) { throw 'Application source paths must be committed and clean before launch.' }
Add-Type -AssemblyName System.Security
$keyPath = Join-Path $taskFolder 'bailian-key.dpapi'
$clearBytes = $null
$taskProcess = $null
$taskStart = $null
$resultCode = 1
try {
  $clearBytes = [Security.Cryptography.ProtectedData]::Unprotect(
    [IO.File]::ReadAllBytes($keyPath), $null,
    [Security.Cryptography.DataProtectionScope]::CurrentUser)
  $taskStart = [Diagnostics.ProcessStartInfo]::new()
  $taskStart.FileName = 'D:/软件安装/node.exe'
  $taskStart.WorkingDirectory = $taskRoot
  $taskStart.UseShellExecute = $false
  $taskStart.CreateNoWindow = $true
  $taskStart.ArgumentList.Add('--import')
  $taskStart.ArgumentList.Add('tsx')
  $taskStart.ArgumentList.Add((Join-Path $taskFolder 'identity-target-diagnostic-20260925.mts'))
  $taskStart.ArgumentList.Add('--quota-verified')
  $taskStart.ArgumentList.Add($QuotaEvidence)
  $taskStart.Environment['IRIS_BAILIAN_EVAL_KEY'] = [Text.Encoding]::UTF8.GetString($clearBytes)
  $taskProcess = [Diagnostics.Process]::Start($taskStart)
  $taskStart.Environment.Remove('IRIS_BAILIAN_EVAL_KEY') | Out-Null
  [Array]::Clear($clearBytes, 0, $clearBytes.Length)
  $clearBytes = $null
  $taskProcess.WaitForExit()
  $resultCode = $taskProcess.ExitCode
} catch {
  Write-Error 'Private identity-target diagnostic launcher failed; no credential details are logged.'
} finally {
  if ($null -ne $taskStart) { $taskStart.Environment.Remove('IRIS_BAILIAN_EVAL_KEY') | Out-Null }
  if ($null -ne $clearBytes) { [Array]::Clear($clearBytes, 0, $clearBytes.Length) }
  if ($null -ne $taskProcess) { $taskProcess.Dispose() }
}
exit $resultCode
