function segment(path: string, value: string, caseId: string): Span[] {
  const spans: Span[] = [];
  let start = 0;
  for (let index = 0; index < value.length; index++) {
    if (!/[。！？\n]/u.test(value[index]!)) continue;
    const end = index + 1;
    const text = value.slice(start, end);
    spans.push({ id: sha(`${caseId}\0${path}\0${start}\0${end}\0${text}`).slice(0, 20),
      path, startUtf16: start, endUtf16: end, text, sha256: sha(text) });
    start = end;
  }
  if (start < value.length) {
    const text = value.slice(start);
    spans.push({ id: sha(`${caseId}\0${path}\0${start}\0${value.length}\0${text}`).slice(0, 20),
      path, startUtf16: start, endUtf16: value.length, text, sha256: sha(text) });
  }
  if (spans.map(item => item.text).join('') !== value
    || spans.some((item, index) => item.startUtf16 !== (index ? spans[index - 1]!.endUtf16 : 0)
      || item.text !== value.slice(item.startUtf16, item.endUtf16) || sha(item.text) !== item.sha256)) {
    throw Error(`Non-exhaustive immutable segmentation: ${caseId}/${path}`);
  }
  return spans;
}

function sanitizedProvider(value: any) {
  const removed: any[] = [];
  const visit = (part: any, path: string): any => {
    if (Array.isArray(part)) return part.map((item, index) => visit(item, `${path}[${index}]`));
    if (part && typeof part === 'object') {
      const clean: Record<string, unknown> = {};
      for (const [key, item] of Object.entries(part)) {
        if (/^(reasoning_content|reasoning_text|thinking_content)$/iu.test(key)) {
          const raw = typeof item === 'string' ? item : JSON.stringify(item);
          removed.push({ path: `${path}.${key}`, bytes: Buffer.byteLength(raw), sha256: sha(raw) });
        } else clean[key] = visit(item, `${path}.${key}`);
      }
      return clean;
    }
    return part;
  };
  return { decoded: visit(value, '$'), removed };
}

        record.failureClassification = 'experimental_output_cap_truncation_not_semantic_quality';
      }
      if (decoded.error || !Array.isArray(decoded.choices) || record.finishReasons.length !== 1
        || record.finishReasons[0] !== 'stop') halt('provider business error or non-stop finish');
      if (halted) throw Error('Provider or usage boundary failed');
      return new Response(bytes, { status: response.status,
        headers: { 'content-type': response.headers.get('content-type') ?? 'application/json' } });
    } catch (error) {
      record.errorType = safeError(error); halt('transport, timeout or response-read failure'); throw error;
    } finally { record.finishedAt = new Date().toISOString(); }
  } }),
});
try {
  for (const item of controls) {
    if (halted || deadline.aborted || report.reportedUsageTokens >= report.maximumReportedUsageTokens) {
      halt('transfer probe stopped before next fixed case'); break;
    }
    active = item;
    const result: any = { controlId: item.id, expectedSupported: item.expectedSupported,
      originalRequestSha256: item.originalRequestSha256,
      immutableSpans: item.spans, originalInput: item.input,
      augmentedMessages: item.messages, augmentedResponseFormat: item.responseFormat,
      startedAt: new Date().toISOString() };
    report.results.push(result);
    try {
      const content = await client.complete(item.messages, { responseFormat: item.responseFormat });
      result.content = redact(content);
      const raw = JSON.parse(content);
      const { claimChecks, validated, allClaimsSupported, conjunction,
        globalConjunctionConsistent } = validateAugmentedReview(raw, item);
      result.schemaValid = true; result.rawSupported = raw.supported;
      result.locallySupported = validated.supported; result.allClaimsSupported = allClaimsSupported;
      result.conjunctionSupported = conjunction; result.validatedProductReview = validated;
      result.claimChecks = claimChecks;
      result.globalConjunctionConsistent = globalConjunctionConsistent;
      if (!result.globalConjunctionConsistent) throw Error('Global supported promoted rejected field or span');
      const byPath = (path: string) => item.spans.filter(span => span.path === path)
        .map(span => ({ ...span, check: claimChecks.find((check: any) => check.id === span.id) }));
      if (item.id === 'old-negative-arithmetic') {
        const wrong = byPath('assessment.reasoning').filter(span => span.text.includes('差额为-6万元'));
        if (wrong.length !== 1) throw Error('Frozen wrong arithmetic sentence unavailable');
        result.targetedChecks = wrong;
        result.targetedProbeMatched = wrong[0].check.supported === false;
      } else if (item.id === 'qualified-risk') {
        const strong = byPath('assessment.reasoning').filter(span => span.text.includes('将直接导致'));
        const extreme = byPath('draft.text').filter(span => span.text.includes('极高'));
        if (strong.length !== 1 || extreme.length !== 1) throw Error('Frozen risk overclaim sentences unavailable');
        result.targetedChecks = [...strong, ...extreme];
        result.targetedProbeMatched = result.targetedChecks.every((span: any) => span.check.supported === false);
      } else if (item.id === 'paraphrase') {
        result.targetedProbeMatched = raw.supported === false
          && raw.fieldChecks.materialChange.supported === false;
      } else {
        result.targetedProbeMatched = validated.supported === true && allClaimsSupported;
      }
      result.fixedOracleBooleanMatched = conjunction === item.expectedSupported;
      if (!result.fixedOracleBooleanMatched || !result.targetedProbeMatched) {
        halt('first fixed boolean or targeted transfer probe mismatch');
      }
    } catch (error) {
      result.schemaValid = false; result.errorType = safeError(error);
      halt('first augmented review schema, local validation or transport failure');
    } finally { result.finishedAt = new Date().toISOString(); }
    if (halted) break;
  }
  report.executionPassed = !halted && report.results.length === 5 && report.requests.length === 5
    && report.results.every((item: any) => item.schemaValid && item.fixedOracleBooleanMatched
      && item.targetedProbeMatched && item.globalConjunctionConsistent);
  if (!report.executionPassed && !halted) halt('five-case transfer probe incomplete');
  process.exitCode = report.executionPassed ? 0 : 1;
} catch (error) {
  report.errorType = safeError(error); halt('transfer probe execution failed'); process.exitCode = 1;
} finally {
  report.halted = halted; report.finishedAt = new Date().toISOString();
  writeFileSync(output, redact(JSON.stringify(report, null, 2)) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ output, requests: report.requests.length,
    reportedUsageTokens: report.reportedUsageTokens, executionPassed: report.executionPassed,
    halted, independentSemanticReview: report.independentSemanticReview }));
}
