function archivedWire(controlId: string) {

  });

}

function validateAnswer(value: any, control: Control) {
  if (!value || typeof value !== 'object' || Array.isArray(value)
    || JSON.stringify(Object.keys(value)) !== JSON.stringify(['verdicts'])
    || !Array.isArray(value.verdicts) || value.verdicts.length !== 3) {
    throw Error('Probe receipt shape invalid');
  }
  const expectedIds = new Set(control.claims.map(claim => claim.id));
  const seen = new Set<string>();
  for (const item of value.verdicts) {
    if (!item || typeof item !== 'object' || Array.isArray(item)
      || JSON.stringify(Object.keys(item).sort()) !== JSON.stringify(['id', 'reason', 'verdict'])
      || typeof item.id !== 'string' || !expectedIds.has(item.id) || seen.has(item.id)
      || !['supported', 'contradicted', 'undetermined'].includes(item.verdict)
      || typeof item.reason !== 'string' || !item.reason.trim() || item.reason.length > 160
      || (item.reason.match(/[。！？!?]/gu) ?? []).length > 1) {
      throw Error('Probe receipt ID, verdict or one-sentence reason invalid');
    }
    seen.add(item.id);
  }
  if (seen.size !== expectedIds.size) throw Error('Probe receipt omitted a fixed ID');
  return value.verdicts.every((item: any) => item.verdict === control.expected[item.id]);
}
function sanitizedProvider(value: any) {
  const removed: any[] = [];
  const visit = (part: any, path: string): any => {
    if (Array.isArray(part)) return part.map((item, index) => visit(item, `${path}[${index}]`));
    if (part && typeof part === 'object') {
      const clean: Record<string, unknown> = {};
      for (const [key, item] of Object.entries(part)) {
        if (/^(reasoning_content|reasoning_text|thinking_content)$/iu.test(key)) {
          const raw = typeof item === 'string' ? item : JSON.stringify(item);
          removed.push({ path: `${path}.${key}`, bytes: Buffer.byteLength(raw), sha256: sha(raw) });
        } else clean[key] = visit(item, `${path}.${key}`);
      }
      return clean;
    }
    return part;
  };
  return { decoded: visit(value, '$'), removed };
}
