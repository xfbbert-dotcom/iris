import assert from 'node:assert/strict';

type Anchor = { quote: string; occurrence: number | null };
type Span = { start: number; end: number };
type Condition = { connective: 'none' | 'single' | 'all' | 'any'; operatorAnchor: Anchor | null; clauses: Anchor[] };
type Unit = { id: string; anchor: Anchor; speechAct: 'assertion' | 'proposal' | 'question';
  epistemicStrength: 'certain' | 'possible' | 'unspecified' | 'not_asserted';
  normativeForce: 'required' | 'recommended' | 'permitted' | 'none'; polarity: 'positive' | 'negative' | 'unspecified';
  operatorAnchors: Anchor[]; parentId: string | null; condition: Condition };
type EventEndpoint = { ownerId: string; eventAnchor: Anchor };
export type Meaning = { coverage: string[]; units: Unit[];
  relations: { kind: 'causes' | 'precedes' | 'and' | 'or'; from: EventEndpoint; to: EventEndpoint; operatorAnchor: Anchor }[];
  referents: { ownerId: string; mention: Anchor; stance: 'assumed_existing' | 'proposed_new' | 'explicit_hypothesis' | 'questioned' | 'generic'; relationAnchor: Anchor }[] };

const object = (value: unknown, keys: readonly string[], name: string): Record<string, any> => {
  assert(value !== null && typeof value === 'object' && !Array.isArray(value), `${name} object`);
  assert.deepEqual(Object.keys(value).sort(), [...keys].sort(), `${name} keys`);
  return value as Record<string, any>;
};
const list = (value: unknown, min: number, max: number, name: string): any[] => {
  assert(Array.isArray(value) && value.length >= min && value.length <= max, `${name} array bound`); return value;
};
const oneOf = (value: unknown, choices: readonly string[], name: string) => assert(typeof value === 'string' && choices.includes(value), `${name} enum`);
const identifier = (value: unknown) => assert(typeof value === 'string' && /^[A-Za-z][A-Za-z0-9_-]{0,19}$/u.test(value), 'identifier');
const inside = (inner: Span, outer: Span) => inner.start >= outer.start && inner.end <= outer.end;

/** Exact anchors and graph integrity are syntax checks, not semantic fidelity proof. */
export function resolveAnchor(value: unknown, claim: string): Span {
  const { quote, occurrence } = object(value, ['quote', 'occurrence'], 'anchor');
  assert(typeof quote === 'string' && quote.trim().length > 0 && quote.length <= 1200, 'anchor quote');
  assert(occurrence === null || Number.isSafeInteger(occurrence) && occurrence >= 1 && occurrence <= 1200, 'anchor occurrence');
  const positions: number[] = [];
  for (let offset = claim.indexOf(quote); offset !== -1; offset = claim.indexOf(quote, offset + 1)) positions.push(offset);
  assert(positions.length > 0, 'anchor quote absent');
  if (occurrence === null) assert.equal(positions.length, 1, 'anchor quote ambiguous; explicit occurrence required');
  const start = positions[(occurrence ?? 1) - 1];
  assert(start !== undefined, 'anchor occurrence unavailable');
  return { start, end: start + quote.length };
}

export function validateMeaning(value: unknown, claim: string): Meaning {
  assert(typeof claim === 'string' && claim.length > 0 && claim.length <= 1200, 'claim bound');
  const graph = object(value, ['coverage', 'units', 'relations', 'referents'], 'meaning');
  const coverage = list(graph.coverage, 1, 24, 'coverage');
  assert(coverage.every(part => typeof part === 'string' && part.length > 0 && part.length <= 1200), 'coverage text');
  assert.equal(coverage.join(''), claim, 'coverage must reproduce unchanged complete claim');
  const units = list(graph.units, 1, 16, 'units');
  const byId = new Map<string, Unit>(); const spans = new Map<string, Span>();
  for (const entry of units) {
    const unit = object(entry, ['id','anchor','speechAct','epistemicStrength','normativeForce','polarity','operatorAnchors','parentId','condition'], 'unit') as Unit;
    identifier(unit.id); assert(!byId.has(unit.id), 'duplicate unit id');
    oneOf(unit.speechAct, ['assertion','proposal','question'], 'speechAct');
    oneOf(unit.epistemicStrength, ['certain','possible','unspecified','not_asserted'], 'epistemicStrength');
    oneOf(unit.normativeForce, ['required','recommended','permitted','none'], 'normativeForce');
    oneOf(unit.polarity, ['positive','negative','unspecified'], 'polarity');
    if (unit.speechAct === 'assertion') {
      assert(unit.epistemicStrength !== 'not_asserted' && unit.normativeForce === 'none', 'speech act assertion consistency');
    } else {
      assert.equal(unit.epistemicStrength, 'not_asserted', 'speech act non-assertion consistency');
      if (unit.speechAct === 'question') assert.equal(unit.normativeForce, 'none', 'speech act question consistency');
    }
    if (unit.parentId !== null) identifier(unit.parentId);
    const span = resolveAnchor(unit.anchor, claim);
    for (const anchor of list(unit.operatorAnchors, 0, 8, 'operators')) {
      assert(inside(resolveAnchor(anchor, claim), span), 'operator outside unit scope');
    }
    const condition = object(unit.condition, ['connective','operatorAnchor','clauses'], 'condition');
    oneOf(condition.connective, ['none','single','all','any'], 'condition');
    const clauses = list(condition.clauses, 0, 8, 'condition clauses');
    assert(condition.connective === 'none' ? clauses.length === 0 && condition.operatorAnchor === null
      : condition.connective === 'single' ? clauses.length === 1 && condition.operatorAnchor !== null
      : clauses.length >= 2 && condition.operatorAnchor !== null, 'condition cardinality');
    for (const anchor of clauses) assert(inside(resolveAnchor(anchor, claim), span), 'condition outside unit scope');
    if (condition.operatorAnchor !== null) assert(inside(resolveAnchor(condition.operatorAnchor, claim), span), 'condition operator outside unit scope');
    byId.set(unit.id, unit); spans.set(unit.id, span);
  }
  for (const unit of byId.values()) {
    if (unit.parentId !== null) {
      assert(byId.has(unit.parentId), 'unknown parent');
      assert(inside(spans.get(unit.id)!, spans.get(unit.parentId)!), 'parent scope does not contain child');
    }
    const seen = new Set<string>(); let id: string | null = unit.id;
    while (id !== null) { assert(!seen.has(id), 'parent cycle'); seen.add(id); id = byId.get(id)!.parentId; }
  }
  const rootSpans = [...byId.values()].filter(unit => unit.parentId === null).map(unit => spans.get(unit.id)!);
  for (let index = 0; index < claim.length; index++) {
    if (/[\p{L}\p{N}]/u.test(claim[index]!)) assert(rootSpans.some(span => index >= span.start && index < span.end), 'unit coverage omits substantive text');
  }
  for (const entry of list(graph.relations, 0, 24, 'relations')) {
    const relation = object(entry, ['kind','from','to','operatorAnchor'], 'relation');
    oneOf(relation.kind, ['causes','precedes','and','or'], 'relation');
    const endpointSpan = (value: unknown) => {
      const endpoint = object(value, ['ownerId','eventAnchor'], 'event endpoint');
      assert(byId.has(endpoint.ownerId), 'relation endpoint reference');
      const eventSpan = resolveAnchor(endpoint.eventAnchor, claim);
      assert(inside(eventSpan, spans.get(endpoint.ownerId)!), 'event endpoint outside owner scope');
      return eventSpan;
    };
    const from = endpointSpan(relation.from), to = endpointSpan(relation.to);
    if (relation.kind === 'causes' || relation.kind === 'precedes') {
      // Owners are linguistic scopes. Overlapping/containing endpoint text cannot
      // turn an entire scope into an event preceding/causing its own subspan.
      assert(from.end <= to.start || to.end <= from.start, 'event endpoints overlap or use a container as its own cause/predecessor');
    }
    resolveAnchor(relation.operatorAnchor, claim);
  }
  for (const entry of list(graph.referents, 0, 24, 'referents')) {
    const referent = object(entry, ['ownerId','mention','stance','relationAnchor'], 'referent');
    assert(byId.has(referent.ownerId), 'referent owner');
    oneOf(referent.stance, ['assumed_existing','proposed_new','explicit_hypothesis','questioned','generic'], 'referent stance');
    assert(inside(resolveAnchor(referent.mention, claim), spans.get(referent.ownerId)!)
      && inside(resolveAnchor(referent.relationAnchor, claim), spans.get(referent.ownerId)!), 'referent outside owner scope');
  }
  return value as Meaning;
}

const str = { type:'string', minLength:1, maxLength:1200 };
const id = { type:'string', minLength:1, maxLength:20, pattern:'^[A-Za-z][A-Za-z0-9_-]{0,19}$' };
const enumeration = (...values: string[]) => ({ type:'string', enum:values });
const obj = (properties: Record<string, unknown>) => ({ type:'object', additionalProperties:false, required:Object.keys(properties), properties });
const arr = (items: unknown, maxItems: number, minItems = 0) => ({ type:'array', minItems, maxItems, items });
const anchor = obj({ quote:str, occurrence:{ anyOf:[{ type:'integer', minimum:1, maximum:1200 }, { type:'null' }] } });
const eventEndpoint = obj({ ownerId:id, eventAnchor:anchor });
export const meaningFormat = { type:'json_schema' as const, json_schema:{ name:'iris_claim_meaning_extraction_v2', strict:true,
  schema:obj({ coverage:arr(str,24,1), units:arr(obj({ id, anchor,
    speechAct:enumeration('assertion','proposal','question'), epistemicStrength:enumeration('certain','possible','unspecified','not_asserted'),
    normativeForce:enumeration('required','recommended','permitted','none'), polarity:enumeration('positive','negative','unspecified'),
    operatorAnchors:arr(anchor,8), parentId:{ anyOf:[id,{ type:'null' }] },
    condition:obj({ connective:enumeration('none','single','all','any'), operatorAnchor:{ anyOf:[anchor,{ type:'null' }] }, clauses:arr(anchor,8) }),
  }),16,1), relations:arr(obj({ kind:enumeration('causes','precedes','and','or'), from:eventEndpoint, to:eventEndpoint, operatorAnchor:anchor }),24),
  referents:arr(obj({ ownerId:id, mention:anchor, stance:enumeration('assumed_existing','proposed_new','explicit_hypothesis','questioned','generic'), relationAnchor:anchor }),24),
  }) } };
