function stripReasoning(value: unknown, path = '$'): { decoded: unknown; removed: any[] } {
  const removed: any[] = [];
  const visit = (part: any, at: string): any => {
    if (Array.isArray(part)) return part.map((item, index) => visit(item, `${at}[${index}]`));
    if (part && typeof part === 'object') {
      const clean: Record<string, unknown> = {};
      for (const [key, item] of Object.entries(part)) {
        if (/^(reasoning_content|reasoning_text|thinking_content)$/iu.test(key)) {
          const raw = typeof item === 'string' ? item : JSON.stringify(item);
          removed.push({ path: `${at}.${key}`, bytes: Buffer.byteLength(raw), sha256: sha(raw) });
        } else clean[key] = visit(item, `${at}.${key}`);
      }
      return clean;
    }
    return part;
  };
  return { decoded: visit(value, path), removed };
}
