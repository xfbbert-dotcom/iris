import { createHash } from 'node:crypto';

import { createOpenAICompatibleChatCompletionsClient } from 'file:///D:/work/AGE-org/.worktrees/iris-daily-pilot-1eb86/apps/core/src/model/openai-compatible-chat-completions-client.ts';

const root = 'D:/work/AGE-org/.worktrees/iris-daily-pilot-1eb86';
const output = 'C:/Users/59912/AppData/Local/Iris/local-eval-20260917/contract-' + phase + '.json';
if (existsSync(output)) throw Error('Existing evidence: no request made');
const config = { provider: 'openai-compatible' as const, baseUrl: 'http://127.0.0.1:18089/v1', apiKey: 'local-synthetic-only', model: 'iris-qwen3-4b-q4km', timeoutMs: 60000 };
const testCase = createProactiveDiscussionEvalCases().find(c => c.id === 'arithmetic')!;
const report: any = {
  kind: 'same-local-model-contract-diagnostic', phase, syntheticOnly: true,
  startedAt: new Date().toISOString(), commit: execFileSync('git', ['rev-parse', 'HEAD'], { cwd: root, encoding: 'utf8' }).trim(),
  modelSourceSha256: createHash('sha256').update(readFileSync(root + '/apps/core/src/proactive-discussion/model.ts')).digest('hex'),
  runtime: 'llama.cpp b11011', model: 'Qwen3-4B-Q4_K_M', seed: 1709,
  timeoutMs: 60000, wallMs: 600000, maxRequests: 14, requestIntervalMs: 6000,
  case: testCase, requests: [], completions: [], manualReview: 'pending',
};
const deadline = AbortSignal.timeout(report.wallMs);
let experiment = 'direct';
const boundedFetch: typeof fetch = async (url, init) => {
  if (String(url) !== config.baseUrl + '/chat/completions' || report.requests.length >= report.maxRequests || deadline.aborted) throw Error('Local diagnostic boundary');
  const payload = JSON.parse(String(init?.body));
  if (payload.model !== config.model || payload.tools || payload.stream) throw Error('Local model boundary');
  payload.seed = 1709;
  const record: any = { index: report.requests.length + 1, experiment, startedAt: new Date().toISOString(), stage: payload.response_format?.json_schema?.name ?? 'plain' };
  report.requests.push(record);
  console.log(JSON.stringify(record));
  try {
    const response = await fetch(url, { ...init, body: JSON.stringify(payload), redirect: 'error', signal: AbortSignal.any([deadline, ...(init?.signal ? [init.signal] : [])]) });
    record.status = response.status;
    return response;
  } finally { record.finishedAt = new Date().toISOString(); }
};
const base = createOpenAICompatibleChatCompletionsClient({ config, fetch: createRequestPacedFetch({ fetch: boundedFetch, requestIntervalMs: 6000 }) });
const client = { async complete(messages: any, options?: any) {
  const entry: any = { experiment, messages, responseFormat: options?.responseFormat, startedAt: new Date().toISOString() };
  report.completions.push(entry);
  try { entry.content = await base.complete(messages, options); return entry.content; }
  catch (error) { entry.error = error instanceof Error ? error.message.slice(0, 500) : 'unknown'; throw error; }
  finally { entry.finishedAt = new Date().toISOString(); }
} };
try {
  report.direct = await client.complete([
    { role: 'system', content: '你是团队协作助手。只根据给出的材料简短回答，用自然中文给出可核对的理由和建议。' },
    { role: 'user', content: testCase.context.items.map(i => i.text).join('\n') + '\n这项预算判断是否有问题？' },

    experiment = 'isolated-render-from-hand-authored-valid-assessment';
    const refs = testCase.context.sources.map(s => s.ref);
    const assessment = {
      decision: 'intervene' as const, reason: 'material_issue' as const,
      issueRef: { kind: 'new' as const, description: '当前招聘方案的成本超过所述预算。' },

      suggestion: '建议核对预算口径，再调整招聘计划或讨论追加预算。', uncertainty: 'fact' as const,
      materialChange: { kind: 'new_issue' as const, explanation: '当前方案的成本合计16万元，与10万元预算相差6万元。', evidenceRefs: refs },
    };
    report.isolatedInput = assessment;
    report.isolatedResult = await createPdModel({ client }).render({ context: testCase.context, assessment });
  }
  process.exitCode = result.error !== null || result.assessment?.decision !== 'intervene' || result.draft === null ? 1 : 0;
} catch (error) {
  report.error = error instanceof Error ? error.message.slice(0, 500) : 'unknown';
  process.exitCode = 1;
} finally {
  report.finishedAt = new Date().toISOString();
  writeFileSync(output, JSON.stringify(report, null, 2) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ output, requests: report.requests.length, results: report.results, direct: report.direct, isolatedResult: report.isolatedResult, error: report.error, exitCode: process.exitCode }));
}
