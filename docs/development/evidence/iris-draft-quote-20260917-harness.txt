type NumberReceipt = { label: string; expectedValue: string; unit: string; draftQuote: string | null };
type Review = { supported: boolean; reason: string; requiredNumbers: NumberReceipt[]; adviceQuote: string | null };
type Target = { expectedValue: string; unit: string };

function numberQuoteValid(draft: string, number: NumberReceipt): boolean {
  if (!isNumberReceipt(number) || number.draftQuote === null) return false;
  const quote = number.draftQuote;
  // Match complete tokens in the full draft: '6万元' is also a substring of '16万元'.
  const tokens = [...draft.matchAll(/(?:[+\-−－负]\s*)?\p{Nd}+(?:[.,，．]\p{Nd}+)*(?:[eE][+\-]?\p{Nd}+)?/gu)];
  for (let offset = draft.indexOf(quote); offset !== -1; offset = draft.indexOf(quote, offset + 1)) {
    for (const token of tokens) {
      if (token[0] !== number.expectedValue) continue;
      const start = token.index!;
      const numberEnd = start + token[0].length;
      const gap = /^[ \t]*/u.exec(draft.slice(numberEnd))![0].length;
      const unitStart = numberEnd + gap;
      const end = unitStart + number.unit.length;
      if (start >= offset && end <= offset + quote.length && draft.startsWith(number.unit, unitStart)) return true;
    }
  }
  return false;
}

function receiptValid(value: unknown, draft: string): boolean {
  if (!isRecord(value) || !hasKeys(value, ['supported', 'reason', 'requiredNumbers', 'adviceQuote'])
    || typeof value.supported !== 'boolean'
    || typeof value.reason !== 'string' || value.reason.trim().length === 0 || value.reason.length > 400
    || !Array.isArray(value.requiredNumbers) || value.requiredNumbers.length > 8
    || typeof value.adviceQuote !== 'string' || value.adviceQuote.trim().length === 0 || value.adviceQuote.length > 1200
    || !draft.includes(value.adviceQuote)) return false;
  return value.requiredNumbers.every(number => isNumberReceipt(number) && numberQuoteValid(draft, number));
}

function expectedCoverage(value: unknown, draft: string, targets: readonly Target[]) {
  const numbers = isRecord(value) && Array.isArray(value.requiredNumbers) ? value.requiredNumbers : [];
  const coverage = targets.map(target => {
    const matches = numbers.filter((number): number is NumberReceipt => isNumberReceipt(number)
      && number.expectedValue === target.expectedValue && number.unit === target.unit);
    return { ...target, declared: matches.length > 0, quotedInDraft: matches.some(number => numberQuoteValid(draft, number)) };
  });
  return { complete: coverage.every(target => target.declared && target.quotedInDraft), targets: coverage };
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null && !Array.isArray(value);
}

function hasKeys(value: Record<string, unknown>, keys: readonly string[]): boolean {
  return Object.keys(value).length === keys.length && keys.every(key => Object.hasOwn(value, key));
}

function isNumberReceipt(value: unknown): value is NumberReceipt {
  return isRecord(value) && hasKeys(value, ['label', 'expectedValue', 'unit', 'draftQuote'])
    && typeof value.label === 'string' && value.label.trim().length > 0 && value.label.length <= 120
    && typeof value.expectedValue === 'string' && /^-?(?:0|[1-9]\d*)(?:\.\d+)?$/u.test(value.expectedValue) && value.expectedValue.length <= 40
    && typeof value.unit === 'string' && value.unit.trim().length > 0 && value.unit === value.unit.trim() && value.unit.length <= 30
    && (value.draftQuote === null || typeof value.draftQuote === 'string' && value.draftQuote.length > 0 && value.draftQuote.length <= 1200);
}

function selfTest() {
  const number = (draftQuote: string | null, expectedValue = '6', unit = '万元'): NumberReceipt => ({ label: '差额', expectedValue, unit, draftQuote });
  const examples: Array<[string, NumberReceipt, boolean]> = [
    ['总额16万元，超预算6万元。', number('超预算6万元'), true],
    ['总额16万元，超预算6万元。', number('6万元'), true],
    ['总额16万元。', number('6万元'), false],
    ['总额16万元。', number('总额16万元'), false],
    ['差额0.6万元。', number('6万元'), false],
    ['差额-6万元。', number('6万元'), false],
    ['差额负6万元。', number('6万元'), false],
    ['差额- 6万元。', number('6万元'), false],
    ['差额1,006万元。', number('6万元'), false],
    ['差额1e6万元。', number('6万元'), false],
    ['差额６6万元。', number('6万元'), false],
    ['差额6.0万元。', number('6万元'), false],
    ['差额6 万元。', number('6 万元'), true],
    ['差额6万元。', number('6'), false],
    ['差额6万元。', number('预算6万元'), false],
    ['差额6万元。', number(null), false],
    ['访谈5人都喜欢。', number('访谈5人', '5', '人'), true],
  ];
  for (const [draft, candidate, expected] of examples) {
    assert.equal(numberQuoteValid(draft, candidate), expected, JSON.stringify({ draft, candidate, expected }));
  }
  const draft = '总额16万元，超预算6万元，建议核对预算。';
  const review: Review = { supported: true, reason: '最终结论。', requiredNumbers: [number('超预算6万元')], adviceQuote: '建议核对预算' };
  assert.equal(receiptValid(review, draft), true);
  assert.equal(receiptValid({ ...review, adviceQuote: '建议增加预算' }, draft), false);
  assert.equal(receiptValid({ ...review, requiredNumbers: [number('16万元')] }, draft), false);
  assert.equal(receiptValid({ ...review, requiredNumbers: [number(null)] }, draft), false);
  assert.equal(receiptValid({ ...review, extra: 'unrequested' }, draft), false);
  assert.equal(receiptValid({ ...review, supported: false }, draft), true);
  const noNumbers = { ...review, requiredNumbers: [] };
  assert.equal(receiptValid(noNumbers, draft), true);
  assert.equal(expectedCoverage(noNumbers, draft, [{ expectedValue: '6', unit: '万元' }]).complete, false);
  assert.equal(noNumbers.supported, true);
  const pairs = loadProbeCases();
  assert.equal(numberQuoteValid(pairs[0]!.input.draft.text, number('6万元')), false);
  assert.equal(numberQuoteValid(pairs[1]!.input.draft.text, number('16 万元', '16')), true);
  assert.equal(numberQuoteValid(pairs[1]!.input.draft.text, number('6 万元')), true);
  assert.equal(numberQuoteValid(pairs[2]!.input.draft.text, number('访谈5人', '5', '人')), true);
  console.log(JSON.stringify({ mode: 'self-test', passed: examples.length + 13, requests: 0, oracleIndependent: true }));
}

function loadProbeCases() {
  const cases = createProactiveDiscussionEvalCases();
  const full = JSON.parse(readFileSync(fullPath, 'utf8'));
  const thinking = JSON.parse(readFileSync(thinkingPath, 'utf8'));
  return [
    { id: 'arithmetic-missing-gap', caseId: 'arithmetic', pair: thinking.results[0], sourcePath: thinkingPath,
      expectedNumbers: [{ expectedValue: '16', unit: '万元' }, { expectedValue: '6', unit: '万元' }] },
    { id: 'arithmetic-complete', caseId: 'arithmetic', pair: full.results.find((result: any) => result.caseId === 'arithmetic' && result.round === 2), sourcePath: fullPath,
      expectedNumbers: [{ expectedValue: '16', unit: '万元' }, { expectedValue: '6', unit: '万元' }] },
    { id: 'inference-wrong-category', caseId: 'inference', pair: full.results.find((result: any) => result.caseId === 'inference' && result.round === 2), sourcePath: fullPath,
      expectedNumbers: [{ expectedValue: '5', unit: '人' }] },
  ].map(({ pair, expectedNumbers, ...entry }) => {
    const sourceCase = cases.find(candidate => candidate.id === entry.caseId);
    if (!sourceCase || pair?.caseId !== entry.caseId || !isRecord(pair.assessment)
      || !isRecord(pair.draft) || typeof pair.draft.text !== 'string') throw Error('Saved synthetic pair is unavailable');
    return { ...entry, round: pair.round, expectedNumbers, input: {
      evidence: sourceCase.context.items.map(item => ({ ...item, kind: 'message' })),
      assessment: pair.assessment, draft: pair.draft as { text: string; evidenceRefs: string[] },
    } };
  });
}

async function run() {
  if (existsSync(output)) throw Error('Probe report already exists; no request made');
  const probeCases = loadProbeCases();
  const config = { provider: 'openai-compatible' as const, baseUrl: 'http://127.0.0.1:18089/v1', apiKey: 'local-synthetic-only',
    model: 'iris-qwen3-4b-q4km', timeoutMs: 180000, structuredOutputMode: 'json_schema' as const };
  const hash = (path: string) => createHash('sha256').update(readFileSync(path)).digest('hex');
  const report: any = {
    kind: 'draft-quote-review-proof-of-concept', syntheticOnly: true, throwawayExperiment: true,
    applicationCommit: 'a5e5e2e426ad0a6cb941770226b313417146762a', originalPairBaseline: '9e4d9ea6',
    productImplementationChangedByProbe: false, manualReview: 'pending',
    startedAt: new Date().toISOString(), model: 'Qwen3-4B-Q4_K_M', runtime: 'llama.cpp b11011', seed: 1709,
    timeoutMs: 180000, totalDeadlineMs: 900000, maxRequests: 3, automaticRetry: false, structuredOutputMode: 'json_schema',
    expectedRuntimeProfile: 'reasoning off; temperature0.7/top-p0.8/top-k20/min-p0/max_tokens2048',
    sourceHashes: { script: hash(folder + '/draft-quote-review-probe.mts'), full: hash(fullPath), thinking: hash(thinkingPath),
      client: hash(root + '/apps/core/src/model/openai-compatible-chat-completions-client.ts') },
    reviewSystem, responseFormat, requests: [], results: [],
    interpretation: {
      exitCode: '0 means all three final contents were received, not that parsing, semantics or receipt checks passed.',
      supported: 'Unmodified model boolean; semantic correctness remains separately reviewed.',
      receiptValid: 'Only returned shape and literal current-draft quote evidence, never the synthetic oracle. Null numeric/advice quotes are not affirmative receipts; an empty numeric list can pass vacuously.',
      expectedCoverage: 'Separate fixed synthetic acceptance oracle; never included in model messages or receiptValid. No numeric/unit equivalence conversion.',
    },
  };
  const deadline = AbortSignal.timeout(900000);
  let activeId: string | null = null;
  const attempted = new Set<string>();
  const client = createOpenAICompatibleChatCompletionsClient({ config,
    sleep: async () => { throw Error('Automatic retry disabled for this probe'); },
    fetch: async (url, init) => {
      if (String(url) !== config.baseUrl + '/chat/completions' || activeId === null
        || attempted.has(activeId) || report.requests.length >= 3 || deadline.aborted) throw Error('Probe request boundary');
      const payload = JSON.parse(String(init?.body));
      if (payload.model !== config.model || payload.tools || payload.stream
        || payload.response_format?.type !== 'json_schema') throw Error('Invalid local probe request');
      attempted.add(activeId);
      const request: any = { caseId: activeId, startedAt: new Date().toISOString() };
      report.requests.push(request);
      try {
        const response = await fetch(url, { ...init, body: JSON.stringify({ ...payload, seed: 1709 }), redirect: 'error',
          signal: AbortSignal.any([deadline, ...(init?.signal ? [init.signal] : [])]) });
        request.status = response.status;
        return response;
      } finally { request.finishedAt = new Date().toISOString(); }
    },
  });
  try {
    for (const probe of probeCases) {
      if (deadline.aborted) break;
      activeId = probe.id;
      const entry: any = { id: probe.id, caseId: probe.caseId, round: probe.round, sourcePath: probe.sourcePath,
        input: probe.input, finalContent: null, supported: null, receiptValid: false, expectedCoverage: null };
      report.results.push(entry);
      try {
        // expectedNumbers is deliberately absent: the oracle must not teach the reviewer its answer.
        entry.finalContent = await client.complete([
          { role: 'system', content: reviewSystem }, { role: 'user', content: JSON.stringify(probe.input) },
        ], { responseFormat });
        let parsed: unknown;
        try { parsed = JSON.parse(entry.finalContent); }
        catch { entry.parseError = 'invalid_json'; }
        entry.supported = isRecord(parsed) && typeof parsed.supported === 'boolean' ? parsed.supported : null;
        entry.receiptValid = receiptValid(parsed, probe.input.draft.text);
        entry.expectedCoverage = expectedCoverage(parsed, probe.input.draft.text, probe.expectedNumbers);
      } catch (error) { entry.error = error instanceof Error ? error.message : 'unknown'; }
      console.log(JSON.stringify({ id: entry.id, supported: entry.supported, receiptValid: entry.receiptValid,
        expectedCoverage: entry.expectedCoverage, error: entry.error }));
    }
  } finally {
    activeId = null;
    report.finishedAt = new Date().toISOString();
    writeFileSync(output, JSON.stringify(report, null, 2) + '\n', { flag: 'wx' });
    console.log(JSON.stringify({ output, requests: report.requests.length, manualReview: 'pending' }));
  }
  process.exitCode = report.results.length === 3 && report.results.every((entry: any) => entry.finalContent !== null && !entry.error) ? 0 : 1;
}

if (process.argv.length !== 3) throw Error('Use exactly --self-test or --run');
if (process.argv[2] === '--self-test') selfTest();
else if (process.argv[2] === '--run') await run();
else throw Error('Use exactly --self-test or --run');
