  'immutable-claim-probe-20260926.halted.json')) {

  throw 'Frozen Qwen3.8-Max review archive unavailable or changed.'

$taskProcess = $null
$taskStart = $null
$resultCode = 1
try {
  $clearBytes = [Security.Cryptography.ProtectedData]::Unprotect(
    [IO.File]::ReadAllBytes($keyPath), $null,
    [Security.Cryptography.DataProtectionScope]::CurrentUser)
