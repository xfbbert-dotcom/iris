function uniqueSentinel(texts: string[], allowed: string) {
  for (const text of texts) for (const size of [40, 32, 24, 16]) {
    for (let start = 0; start + size <= text.length; start++) {
      const span = text.slice(start, start + size);
      if (span.trim().length >= 12 && !allowed.includes(span)) return span;
    }
  }
  throw Error('No unique other-artifact sentinel; isolation cannot be shown');
}

function validateDraft(value: any, draftText: string) {
  if (!strictKeys(value, ['supported', 'reason', 'requiredNumbers', 'adviceQuote'])

    || !value.reason.trim() || value.reason.length > 200

    throw Error('Document-only receipt shape invalid');
  }
  if (value.requiredNumbers.some((entry: any) => !strictKeys(entry,
    ['label', 'expectedValue', 'unit', 'draftQuote'])
    || typeof entry.label !== 'string' || !entry.label.trim() || entry.label.length > 80
    || typeof entry.expectedValue !== 'string'
    || entry.expectedValue.length > 40
    || !/^-?(0|[1-9][0-9]*)(\.[0-9]+)?$/u.test(entry.expectedValue)
    || typeof entry.unit !== 'string' || !entry.unit.trim() || entry.unit.length > 24
    || !(entry.draftQuote === null || (typeof entry.draftQuote === 'string'
      && entry.draftQuote.length > 0 && entry.draftQuote.length <= 1200)))) {
    throw Error('Document-only number receipt shape invalid');
  }
  const numberQuoteMatches = (entry: any) => {
    if (entry.draftQuote === null) return false;
    const tokens = draftText.matchAll(/(?:[+\-−－负]\s*)?\p{Nd}+(?:[.,，．]\p{Nd}+)*(?:[eE][+\-]?\p{Nd}+)?/gu);
    for (const token of tokens) {
      if (token[0] !== entry.expectedValue) continue;
      const start = token.index!;
      const numberEnd = start + token[0].length;
      const gap = /^[ \t]*/u.exec(draftText.slice(numberEnd))![0].length;
      const unitStart = numberEnd + gap;
      const end = unitStart + entry.unit.length;
      if (!draftText.startsWith(entry.unit, unitStart)) continue;
      const quoteStart = draftText.lastIndexOf(entry.draftQuote, start);
      if (quoteStart !== -1 && end <= quoteStart + entry.draftQuote.length) return true;
    }
    return false;
  };
  const invalidQuote = value.requiredNumbers.some((entry: any) => entry.draftQuote !== null
    && !numberQuoteMatches(entry));
  const invalidAdvice = value.adviceQuote !== null && !draftText.includes(value.adviceQuote);
  if (invalidQuote || invalidAdvice || (value.supported && (value.adviceQuote === null
    || value.requiredNumbers.some((entry: any) => entry.draftQuote === null)))) {
    throw Error('Document-only literal receipt invalid');
  }
  return value.supported as boolean;
}
function sanitizedProvider(value: any) {
  const removed: any[] = [];
  const visit = (part: any, path: string): any => {
    if (Array.isArray(part)) return part.map((item, index) => visit(item, `${path}[${index}]`));
    if (part && typeof part === 'object') {
      const clean: Record<string, unknown> = {};
      for (const [key, item] of Object.entries(part)) {
        if (/^(reasoning_content|reasoning_text|thinking_content)$/iu.test(key)) {
          const raw = typeof item === 'string' ? item : JSON.stringify(item);
          removed.push({ path: `${path}.${key}`, bytes: Buffer.byteLength(raw), sha256: sha(raw) });
        } else clean[key] = visit(item, `${path}.${key}`);
      }
      return clean;
    }
    return part;
  };
  return { decoded: visit(value, '$'), removed };
}

    const result: any = { controlId: control.id, stage: control.stage,
      expectedSupported: control.expectedSupported, originalMessages: messages,
      originalResponseFormat: responseFormat, startedAt: new Date().toISOString() };
    report.results.push(result);
    try {
      const content = await client.complete(messages, { responseFormat });
      result.content = redact(content);
      const value = JSON.parse(content);
      const locallySupported = control.stage === 'claims'
        ? validateClaims(value) : validateDraft(value, (control.user.draft as any).text);
      result.schemaValid = true;
      result.rawSupported = value.supported;
      result.locallySupported = locallySupported;
      result.profileVerdictMatched = locallySupported === control.expectedSupported;
      if (!result.profileVerdictMatched) halt('first fixed-control judgement mismatch; probe failed');
    } catch (error) {
      result.schemaValid = false; result.errorType = safeError(error);
      halt('first fixed-control schema, transport or validation failure; probe failed');
    } finally { result.finishedAt = new Date().toISOString(); }
    if (halted) break;
  }
  report.executionPassed = !halted && report.results.length === 6 && report.requests.length === 6
    && report.results.every((item: any) => item.profileVerdictMatched === true);
  if (!report.executionPassed && !halted) halt('six-control probe incomplete');
  process.exitCode = report.executionPassed ? 0 : 1;
} catch (error) {
  report.errorType = safeError(error); halt('probe execution failed'); process.exitCode = 1;
} finally {
  report.halted = halted; report.finishedAt = new Date().toISOString();
  writeFileSync(output, redact(JSON.stringify(report, null, 2)) + '\n', { flag: 'wx' });
  console.log(JSON.stringify({ output, requests: report.requests.length,
    reportedUsageTokens: report.reportedUsageTokens, executionPassed: report.executionPassed,
    halted, independentSemanticReview: report.independentSemanticReview }));
}
